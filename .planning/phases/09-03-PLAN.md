<script lang="ts">
  import { providerState, switchProvider, setApiKey } from '../../stores/llm-provider'
  import { settings, setProviderConfig } from '../../stores/settings'
  import { toast } from '../../services/toast'

  let primaryProvider = $state($settings.suggestions.llmProvider)
  let fallbackChain = $state([...$settings.suggestions.fallbackChain])
  let openrouterApiKey = $state('')
  let showApiKeyInput = $state(false)
  let isSavingApiKey = $state(false)

  async function handleSwitchProvider(provider: 'ollama' | 'openrouter') {
    try {
      await switchProvider(provider, openrouterApiKey || undefined)
      primaryProvider = provider
      setProviderConfig(provider, fallbackChain)
      toast(`Switched to ${provider}`, 'success')
    } catch (err) {
      toast(err instanceof Error ? err.message : 'Failed to switch provider', 'error')
    }
  }

  async function handleSaveApiKey() {
    if (!openrouterApiKey.trim()) {
      toast('API key cannot be empty', 'error')
      return
    }

    isSavingApiKey = true
    try {
      await setApiKey('openrouter', openrouterApiKey)
      toast('OpenRouter API key saved securely', 'success')
      showApiKeyInput = false
    } catch (err) {
      toast(err instanceof Error ? err.message : 'Failed to save API key', 'error')
    } finally {
      isSavingApiKey = false
    }
  }

  function toggleFallback(provider: 'ollama' | 'openrouter') {
    if (fallbackChain.includes(provider)) {
      fallbackChain = fallbackChain.filter((p) => p !== provider)
    } else {
      fallbackChain = [...fallbackChain, provider]
    }
    setProviderConfig(primaryProvider, fallbackChain)
  }
</script>

<script lang="ts">
  import { settings, setTelemetryOptIn } from '../../stores/settings'
  import { getTelemetryStats, clearTelemetry } from '../../stores/telemetry'
  import { toast } from '../../services/toast'

  let telemetryOptIn = $state($settings.suggestions.telemetryOptIn)
  let stats = $state<any>(null)
  let isLoading = $state(false)

  // Load stats when opting in
  async function handleOptInChange(enabled: boolean) {
    telemetryOptIn = enabled
    setTelemetryOptIn(enabled)

    if (enabled) {
      await loadStats()
      toast('Telemetry enabled', 'info')
    } else {
      stats = null
      toast('Telemetry disabled', 'info')
    }
  }

  async function loadStats() {
    isLoading = true
    try {
      stats = await getTelemetryStats()
    } catch (err) {
      toast('Failed to load telemetry stats', 'error')
    } finally {
      isLoading = false
    }
  }

  async function handleClearTelemetry() {
    if (!confirm('Clear all telemetry history? This cannot be undone.')) return

    try {
      await clearTelemetry()
      stats = null
      toast('Telemetry cleared', 'success')
    } catch (err) {
      toast('Failed to clear telemetry', 'error')
    }
  }

  // Refresh stats on mount
  $effect.pre(() => {
    if (telemetryOptIn && !stats) {
      loadStats()
    }
  })
</script>

<div class="telemetry-dashboard">
  <h3>Privacy & Telemetry</h3>

  <div class="setting-group">
    <label>
