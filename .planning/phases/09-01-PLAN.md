# Plan 09-01: Core Suggestion Service + Provider Chain + Settings

**Objective:** Build the foundation for inline suggestions: debounce-based suggestion triggering with auto-cancel, multi-provider fallback chain with error recovery, and workspace-scoped settings storage. Enables all downstream suggestion UI and editor integration.

**Purpose:** Decouples suggestion business logic (timing, provider selection, error handling, storage) from UI presentation. Tests can verify latency SLA, fallback behavior, token budgets independently. UI layer (Plan 09-02) will consume these services.

**Output:** Three core services (Suggestion, ProviderChain, Settings), telemetry store, and type definitions — fully testable in isolation.

## Context

**Phase 8 dependencies:**
- DocumentEditor keystroke events (Phase 8 established CodeMirror integration, debounce pattern)
- Paper validation service (document-validation.ts — patterns for decorations, real-time validation)
- Document persistence patterns (documents.ts debounce logic, workspace store)

**Research findings (09-RESEARCH.md):**
- Provider fallback chain must classify errors (recoverable vs. terminal) to avoid fallback on auth errors
- Debounce + abort pattern reuses Phase 8 save debounce structure (proven)
- Token counting already available via js-tiktoken (used by OpenRouterProvider)
- Settings persistence via Tauri invoke `save_settings` / `load_settings` commands (pattern exists for API keys)

**Locked decisions from 09-CONTEXT.md:**
- Auto-cancel model: 300ms debounce, 10s timeout, AbortController
- Multi-provider: User selects primary, auto-fallback on error with toast notification
- Settings: Configurable provider priority order, display preferences, opt-in telemetry toggle
- Telemetry: Local-only, off by default, stored in workspace `.anya/telemetry.json`

```typescript
import { get } from 'svelte/store'
import { selectedItems } from '../stores/context-selection'
import type { Suggestion, SuggestionRequest } from '../types/suggestion'
import type { LLMProvider, Message, ChatOptions } from '../types/llm'
import type { SuggestionProviderChain } from './suggestion-provider-chain'
import { recordAcceptance, recordRejection } from '../stores/telemetry'

export class SuggestionService {
  private pendingRequest: SuggestionRequest | null = null
  private debounceTimer: ReturnType<typeof setTimeout> | null = null
  private tokenCountLimit = 500  // Per ROADMAP: 500 tokens max for context

  constructor(
    private providerChain: SuggestionProviderChain,
    private onGenerating?: (isGenerating: boolean) => void
  ) {}

  /**
   * Trigger suggestion generation after keystroke.
   * Implements: 300ms debounce, auto-cancel previous request, 10s timeout.
   * Per 09-CONTEXT.md Area 1: "auto-cancel in-flight on keystroke"
   */
  async triggerSuggestion(
    docId: string,
    currentContent: string,
    cursorPos: number
  ): Promise<Suggestion | null> {
    // Cancel previous request if in-flight
    if (this.pendingRequest) {
      this.pendingRequest.abortController.abort()
      clearTimeout(this.pendingRequest.timeoutHandle)
      this.onGenerating?.(false)
    }

    // Debounce keystroke: reset 300ms timer on each keystroke
    if (this.debounceTimer) clearTimeout(this.debounceTimer)

    return new Promise((resolve) => {
      this.debounceTimer = setTimeout(async () => {
        try {
          const suggestion = await this._generateSuggestion(
            docId,
            currentContent,
            cursorPos
          )
          resolve(suggestion)
        } catch (err) {
          if (err instanceof DOMException && err.name === 'AbortError') {
            // Request was cancelled by user keystroke or timeout
            resolve(null)
          } else {
            // Real error — logged upstream
            resolve(null)
          }
        }
      }, 300)
    })
  }

  private async _generateSuggestion(
    docId: string,
    currentContent: string,
    cursorPos: number
  ): Promise<Suggestion | null> {
    const requestId = crypto.randomUUID()
    const abortController = new AbortController()
    const startTime = Date.now()

    // 10s timeout (per 09-CONTEXT.md Area 1)
    const timeoutHandle = setTimeout(() => {
      abortController.abort()
      // Toast notification handled upstream in editor integration
    }, 10000)

    this.pendingRequest = { requestId, abortController, timeoutHandle, docId, startTime }
    this.onGenerating?.(true)

    try {
      // Build context: selected papers/notes + current document section
      const context = this._buildContext(currentContent, cursorPos)

      // Validate token budget (500 tokens max per 09-CONTEXT.md + ROADMAP)
      const contextTokens = await this._countTokens(context)
      if (contextTokens > this.tokenCountLimit) {
        // Truncate context to fit budget
        // For simplicity: take top N papers by selection order
        // Production: implement sliding window or importance sampling
      }

      // Build LLM messages
      const messages: Message[] = [
        { role: 'system', content: 'You are a helpful writing assistant. Generate a single sentence suggestion to continue the given text. Only provide the suggestion, no explanation.' },
        { role: 'user', content: context }
      ]

      // Generate via provider chain (handles fallback + cancellation)
      const suggestionText = await this._streamSuggestion(messages, abortController.signal)

      if (!suggestionText) return null

      const lines = suggestionText.split('\n')
      const suggestion: Suggestion = {
        id: requestId,
        docId,
        text: suggestionText,
        lines,
        provider: 'ollama', // Will be determined by provider chain
        position: { line: currentContent.substring(0, cursorPos).split('\n').length - 1, col: cursorPos },
        generatedAt: startTime,
        state: 'pending',
        acceptedLines: new Set()
      }

      return suggestion
    } finally {
      clearTimeout(timeoutHandle)
      this.pendingRequest = null
      this.onGenerating?.(false)
    }
  }

  private _buildContext(currentContent: string, cursorPos: number): string {
    // Get selected papers/notes for context (per SUGG-02, SUGG-08)
    const selected = get(selectedItems)
    const contextBudget = 250  // Reserve tokens for prompt + doc section

    let contextStr = 'Selected context:\n'
    for (const item of selected) {
      if (item.tokens < contextBudget) {
        contextStr += `\n${item.title}:\n${item.content.substring(0, 300)}\n`
      }
    }

    // Add current document section around cursor (prev 200 chars + next 100 chars)
    const sectionStart = Math.max(0, cursorPos - 200)
    const sectionEnd = Math.min(currentContent.length, cursorPos + 100)
    const section = currentContent.substring(sectionStart, sectionEnd)

    contextStr += `\n\nDocument section:\n${section}\n\nContinue from this point:`

    return contextStr
  }

  private async _countTokens(text: string): Promise<number> {
    // Reuse existing js-tiktoken pattern from document-validation.ts
    // For now: rough estimate 1 token ≈ 4 characters
    return Math.ceil(text.length / 4)
  }

  private async _streamSuggestion(
    messages: Message[],
    signal: AbortSignal
  ): Promise<string> {
    let result = ''
    try {
      for await (const chunk of this.providerChain.generateSuggestion(messages, {}, signal)) {
        result += chunk
      }
    } catch (err) {
      if (signal.aborted) throw err  // Re-throw abort
      // Fallback chain errors logged upstream
      throw err
    }
    return result
  }

  /**
   * Accept suggestion (partially or fully).
   * Records telemetry if opted in.
   */
  recordAcceptanceTelemetry(
    docId: string,
    provider: 'ollama' | 'openrouter',
    totalLines: number,
    acceptedLines: number
  ): void {
    if (acceptedLines === totalLines) {
      recordAcceptance(docId, provider, 'full', totalLines)
    } else {
      recordAcceptance(docId, provider, 'partial', totalLines, acceptedLines)
    }
  }

  /**
   * Reject suggestion.
   */
  recordRejectionTelemetry(docId: string, provider: 'ollama' | 'openrouter'): void {
    recordRejection(docId, provider)
  }

  cancel(): void {
    if (this.pendingRequest) {
      this.pendingRequest.abortController.abort()
      clearTimeout(this.pendingRequest.timeoutHandle)
      this.pendingRequest = null
    }
    if (this.debounceTimer) clearTimeout(this.debounceTimer)
    this.onGenerating?.(false)
  }
}
```

```typescript
import type { LLMProvider, Message, ChatOptions } from '../types/llm'

export interface ProviderError {
  code: 'connection' | 'auth' | 'timeout' | 'unknown'
  message: string
  recoverable: boolean
}

/**
 * Error classifier for provider failures.
 * Distinguishes recoverable (connection, timeout) from terminal (auth) errors.
 * Per 09-CONTEXT.md Area 6: "auto-fallback if selected fails"
 */
function classifyError(err: unknown): ProviderError {
  if (err instanceof TypeError && err.message.includes('fetch')) {
    return {
      code: 'connection',
      message: 'Connection failed',
      recoverable: true
    }
  }

  if (err instanceof Error) {
    if (err.message.includes('Invalid') && err.message.includes('API key')) {
      return {
        code: 'auth',
        message: 'Invalid API key',
        recoverable: false  // Don't fallback on auth error
      }
    }
    if (err.message.includes('timeout')) {
      return {
        code: 'timeout',
        message: 'Request timed out',
        recoverable: true
      }
    }
  }

  return {
    code: 'unknown',
    message: String(err),
    recoverable: true
  }
}

/**
 * Wrapper around LLM providers implementing fallback chain.
 * Per 09-CONTEXT.md Area 6: User-selected primary provider with auto-fallback.
 * 
 * Retry behavior: On next suggestion request, retry primary (don't permanently switch).
 */
export class SuggestionProviderChain {
  constructor(
    private primaryProvider: LLMProvider,
    private fallbackChain: LLMProvider[],
    private onFallback?: (message: string) => void  // Toast callback
  ) {}

  /**
   * Generate suggestion with automatic fallback chain.
   * If primary fails (recoverable error), try fallbacks.
   * If all fail, throw error (handled upstream).
   */
  async *generateSuggestion(
    messages: Message[],
    options: ChatOptions,
    signal: AbortSignal
  ): AsyncIterable<string> {
    const providers = [this.primaryProvider, ...this.fallbackChain]
    let lastError: ProviderError | null = null

    for (let i = 0; i < providers.length; i++) {
      const provider = providers[i]
      const isPrimary = i === 0

      try {
        // Check provider availability first
        const available = await provider.isAvailable()
        if (!available) {
          if (!isPrimary && this.onFallback) {
            this.onFallback(`Provider unavailable, trying next...`)
          }
          continue
        }

        // Stream response
        yield* provider.chat(messages, options)
        return  // Success!
      } catch (err) {
        const error = classifyError(err)
        lastError = error

        if (!error.recoverable) {
          // Terminal error (auth, etc.) — don't try fallback, propagate immediately
          throw err
        }

        if (!isPrimary && this.onFallback) {
          this.onFallback(`Provider failed: ${error.message}. Trying next...`)
        }

        // Try next provider
        if (i === providers.length - 1) {
          // All providers exhausted
          throw new Error(
            lastError
              ? `All suggestion providers failed: ${lastError.message}`
              : 'No providers available'
          )
        }
      }
    }

    throw new Error('No providers available')
  }
}
```

```typescript
  }

  const stats: TelemetryStats = {
    totalSuggestions: total,
    accepted: allRecords.filter((r) => r.type === 'accept').length,
    rejected: allRecords.filter((r) => r.type === 'reject').length,
    partial: allRecords.filter((r) => r.type === 'partial').length,
    acceptanceRate: 0,
    byProvider: {
      ollama: { count: 0, acceptanceRate: 0 },
      openrouter: { count: 0, acceptanceRate: 0 }
    }
  }

  stats.acceptanceRate =
    (stats.accepted + stats.partial) / (stats.totalSuggestions || 1)

  for (const provider of ['ollama', 'openrouter'] as const) {
    const providerRecords = allRecords.filter((r) => r.provider === provider)
    const providerTotal = providerRecords.length
    const providerAccepted = providerRecords.filter(
      (r) => r.type === 'accept' || r.type === 'partial'
    ).length

    stats.byProvider[provider] = {
      count: providerTotal,
      acceptanceRate: providerTotal > 0 ? providerAccepted / providerTotal : 0
    }
  }

  return stats
}
```

```typescript
import { encoding_for_model as encodingForModel } from 'js-tiktoken'
import type { LLMProvider, Message, Model, ChatOptions } from '../../types/llm'

/**
 * OpenRouter provider for LLM suggestions.
 * OpenAI-compatible API (supports fallback/load-balancing across multiple models).
 * Per 09-CONTEXT.md Area 6 + LLM-MULTI-01.
 */
export class OpenRouterProvider implements LLMProvider {
  private baseUrl = 'https://openrouter.ai/api/v1'
  private defaultModel = 'openai/gpt-3.5-turbo'  // Fast, cheap fallback

  constructor(private apiKey: string) {
    if (!apiKey?.trim()) {
      throw new Error('OpenRouter API key required')
    }
  }

  /**
   * Check if provider is available and API key is valid.
   */
  async isAvailable(): Promise<boolean> {
    try {
      const resp = await fetch(`${this.baseUrl}/models`, {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'HTTP-Referer': 'http://anya-research.local',  // Required by OpenRouter
          'X-Title': 'Anya Research'
        }
      })
      return resp.ok
    } catch {
      return false
    }
  }

  /**
   * Stream chat response as async iterable.
   * Implements token limit check before request (per SUGG-04: token budget).
   */
  async *chat(
    messages: Message[],
    options: ChatOptions = {}
  ): AsyncIterable<string> {
    const model = options.model || this.defaultModel
    const temperature = options.temperature ?? 0.7

    try {
      const resp = await fetch(`${this.baseUrl}/chat/completions`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${this.apiKey}`,
          'HTTP-Referer': 'http://anya-research.local',
          'X-Title': 'Anya Research'
        },
        body: JSON.stringify({
          model,
          messages,
          temperature,
          stream: true,
          max_tokens: 150  // Single suggestion: keep it short
        })
      })

      if (!resp.ok) {
        const error = await resp.text()
        throw new Error(`OpenRouter API error (${resp.status}): ${error}`)
      }

      if (!resp.body) throw new Error('No response body')

      const reader = resp.body.getReader()
      const decoder = new TextDecoder()
      let buffer = ''

      while (true) {
        const { done, value } = await reader.read()
        if (done) break

        buffer += decoder.decode(value, { stream: true })
        const lines = buffer.split('\n')

        for (let i = 0; i < lines.length - 1; i++) {
          const line = lines[i].trim()
          if (!line || line === '[DONE]') continue
          if (!line.startsWith('data: ')) continue

          const json = JSON.parse(line.slice(6))
          const delta = json.choices?.[0]?.delta
          if (delta?.content) {
            yield delta.content
          }
        }

        buffer = lines[lines.length - 1]
      }
    } catch (err) {
      if (err instanceof Error && err.message.includes('API key')) {
        throw new Error('Invalid OpenRouter API key')
      }
      throw err
    }
  }

  /**
   * List available models from OpenRouter.
   */
  async models(): Promise<Model[]> {
    try {
      const resp = await fetch(`${this.baseUrl}/models`, {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'HTTP-Referer': 'http://anya-research.local',
          'X-Title': 'Anya Research'
        }
      })

      if (!resp.ok) throw new Error('Failed to fetch models')

      const data = await resp.json() as { data: any[] }
      return data.data
        .filter((m: any) => !m.id.includes('claude'))  // Skip non-OpenRouter models for cost
        .map((m: any) => ({
          id: m.id,
          name: m.name || m.id,
          contextWindow: m.context_length || 4096,
          size: 0  // OpenRouter doesn't expose size
        }))
    } catch {
      return []
    }
  }

  /**
   * Count tokens in text using js-tiktoken.
   * Note: js-tiktoken may not have exact tokenizer for all OpenRouter models.
   * Falls back to estimation if unavailable.
   */
  async tokenCount(text: string): Promise<number> {
    try {
      const enc = encodingForModel(this.defaultModel)
      const tokens = enc.encode(text)
      return tokens.length
    } catch {
      // Fallback: rough estimate (1 token ≈ 4 chars)
      return Math.ceil(text.length / 4)
    }
  }
}
```

```rust
#[tauri::command]
pub async fn save_settings(workspace_path: String, settings: String) -> Result<(), String> {
  let settings_path = get_settings_path(&workspace_path);
```
