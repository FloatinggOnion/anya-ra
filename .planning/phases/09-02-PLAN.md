# Plan 09-02: Editor Integration + Ghost Text Decorations

**Objective:** Integrate suggestion service into DocumentEditor with CodeMirror decorations for line-by-line partial acceptance. Renders suggestions as non-intrusive ghost text without modifying document content until user explicitly accepts.

**Purpose:** Users see real-time writing suggestions flowing through their editor (responsive, low-latency experience). Line-by-line granularity provides control without all-or-nothing commitment. Ghost text decoration pattern integrates with CodeMirror's undo/redo system to preserve atomicity.

**Output:** DocumentEditor keystroke integration with SuggestionService, CodeMirror StateField + decoration for ghost text rendering, Tab/Escape handlers for acceptance/rejection.

```typescript
import { StateField, StateEffect } from '@codemirror/state'
import { Decoration, EditorView, WidgetType } from '@codemirror/view'
import type { Text } from '@codemirror/state'
import type { Suggestion } from '../../types/suggestion'

/**
 * StateEffect to update suggestion text in editor.
 * Fired when suggestion service generates a new suggestion or user accepts/rejects.
 */
export const setSuggestion = StateEffect.define<{
  suggestion: Suggestion | null
  cursorPos: number
}>()

/**
 * StateEffect to toggle acceptance of specific suggestion lines.
 * Fired when user presses Shift+Tab to toggle line acceptance.
 */
export const toggleLineAcceptance = StateEffect.define<number>()

/**
 * CodeMirror WidgetType for rendering ghost text inline.
 * Displays as faint, grayed-out text that doesn't interfere with editing.
 */
export class GhostTextWidget extends WidgetType {
  constructor(readonly text: string) {}

  toDOM(): HTMLElement {
    const span = document.createElement('span')
    span.className = 'suggestion-ghost-text'
    span.textContent = this.text
    span.setAttribute('aria-label', 'Suggestion - press Tab to accept, Escape to reject')
    return span
  }

  ignoreEvent(): boolean {
    return false  // Allow events through
  }

  get estimatedHeight(): number {
    // Return -1 to let CodeMirror measure
    return -1
  }
}

/**
 * Internal state tracked by suggestionField.
 * Separate from document content to preserve undo/redo integrity.
 */
interface SuggestionState {
  suggestion: Suggestion | null
  acceptedLines: Set<number>  // Line indices user accepted (for partial acceptance)
  cursorPos: number  // Position where suggestion was generated
}

/**
 * CodeMirror StateField for managing suggestion state + decorations.
 * Per Phase 9 RESEARCH.md Pattern 2: integrate with transaction system.
 */
export const suggestionField = StateField.define<SuggestionState>({
  create() {
    return {
      suggestion: null,
      acceptedLines: new Set<number>(),
      cursorPos: 0
    }
  },

  update(state, tr) {
    let newState = { ...state }

    for (const effect of tr.effects) {
      if (effect.is(setSuggestion)) {
        // New suggestion received (or cleared if null)
        newState.suggestion = effect.value.suggestion
        newState.cursorPos = effect.value.cursorPos
        newState.acceptedLines = new Set<number>()  // Reset accepted lines
      } else if (effect.is(toggleLineAcceptance)) {
        // User toggled specific line acceptance (Shift+Tab)
        const lineIdx = effect.value
        if (newState.acceptedLines.has(lineIdx)) {
          newState.acceptedLines.delete(lineIdx)
        } else {
          newState.acceptedLines.add(lineIdx)
        }
      }
    }

    // Clear suggestion if content changed significantly (e.g., user typed while suggestion visible)
    // Compare document before/after
    if (tr.docChanged) {
      // If cursor moved or content around suggestion changed, invalidate
      // (The keystroke that triggered content change will debounce a new suggestion anyway)
      // For now: keep suggestion visible — user can press Escape to dismiss
    }

    return newState
  },

  /**
   * Provide decorations for rendering ghost text.
   * Only create decoration for non-accepted lines.
   */
  provide: (f) =>
    EditorView.decorations.compute([f], (state) => {
      const { suggestion, acceptedLines, cursorPos } = state.field(f)

      if (!suggestion) {
        return Decoration.none
      }

      const decorations = []
      const lines = suggestion.lines

      for (let i = 0; i < lines.length; i++) {
        if (!acceptedLines.has(i)) {
          // Create decoration for non-accepted line
          // Insert after cursor position (or at end of current line)
          const pos = state.doc.length > cursorPos ? cursorPos : state.doc.length

          const decoration = Decoration.widget({
            widget: new GhostTextWidget(
              i === 0 ? lines[i] : '\n' + lines[i]
            ),
            side: 1  // Insert after cursor (positive side)
          }).range(pos)

          decorations.push(decoration)
        }
      }

      return Decoration.set(decorations)
    })
})

/**
 * Helper to clear suggestion from editor state.
 * Called when user presses Escape or accepts suggestion.
 */
export function clearSuggestion(view: EditorView): void {
  view.dispatch({
    effects: setSuggestion.of({ suggestion: null, cursorPos: 0 })
  })
}

/**
 * Helper to get accepted lines from suggestion state.
 * Used by acceptance handler to determine which lines to insert.
 */
export function getAcceptedLines(view: EditorView): Set<number> {
  const state = view.state.field(suggestionField)
  return state.acceptedLines
}

/**
 * Helper to get current suggestion from editor state.
 */
export function getCurrentSuggestion(view: EditorView): Suggestion | null {
  const state = view.state.field(suggestionField)
  return state.suggestion
}
```

import { toast } from '../../services/toast'
import type { Suggestion } from '../../types/suggestion'

// ... existing code ...

let suggestionService: SuggestionService | null = null
let isGeneratingSuggestion = $state(false)
let suggestionLoadingTimer: ReturnType<typeof setTimeout> | null = null

// Initialize suggestion service when editor mounts
$effect(() => {
  const editor = editor  // Reference to editor from NotesEditor component
  const currentSettings = $settings
  const providerState = $providerState

  if (!editor || !currentSettings.suggestions.enabled || !providerState.provider) {
    suggestionService = null
    return
  }

  // Create fallback chain based on settings
  const primaryProvider = providerState.provider
  const fallbackProviders = currentSettings.suggestions.fallbackChain
    .filter((p) => p !== currentSettings.suggestions.llmProvider)
    .map(async (p) => {
      if (p === 'ollama') {
        const { OllamaProvider } = await import('../../services/llm/ollama')
        return new OllamaProvider()
      } else {
        const { OpenRouterProvider } = await import('../../services/llm/openrouter')
        const apiKey = providerState.apiKey
        if (!apiKey) throw new Error('OpenRouter API key not configured')
        return new OpenRouterProvider(apiKey)
      }
    })

  const chain = new SuggestionProviderChain(
    primaryProvider,
    await Promise.all(fallbackProviders),
    (msg) => toast(msg, 'info')
  )

  suggestionService = new SuggestionService(chain, (isGenerating) => {
    isGeneratingSuggestion = isGenerating

    // Show loading indicator if >1s (per SUGG-07)
    if (isGenerating && !suggestionLoadingTimer) {
      suggestionLoadingTimer = setTimeout(() => {
        // Toast or visual indicator could be shown here
        console.log('[suggestions] Generation taking >1s')
      }, 1000)
    } else if (!isGenerating && suggestionLoadingTimer) {
      clearTimeout(suggestionLoadingTimer)
      suggestionLoadingTimer = null
    }
  })
})

// Handle keystroke: trigger debounced suggestion + handle acceptance/rejection
async function handleEditorChange(event: any) {
  const newContent = event.detail.state.doc.toString()
  if (newContent === lastLoadedContent) return
  if (!$selectedDocumentId || !$currentDocument) return

  // Update content state
  content = newContent

  // Auto-save (existing)
  if (autoSaveTimeout) clearTimeout(autoSaveTimeout)
  autoSaveTimeout = setTimeout(() => {
    handleAutoSave(newContent)
  }, 300)

  // Trigger suggestion (if enabled)
  if (suggestionService && $settings.suggestions.enabled) {
    const cursorPos = event.detail.state.selection.main.head
    const suggestion = await suggestionService.triggerSuggestion(
      $selectedDocumentId,
      newContent,
      cursorPos
    )

    if (suggestion) {
      // Update CodeMirror decoration with suggestion
      const editor = event.detail.view
      editor.dispatch({
        effects: setSuggestion.of({
          suggestion,
          cursorPos
        })
      })
    }
  }
}

// Handle Tab/Escape for suggestion acceptance/rejection
function handleEditorKeydown(event: KeyboardEvent) {
  const editor = editor  // From NotesEditor ref
  if (!editor) return

  const suggestion = getCurrentSuggestion(editor)
  if (!suggestion) return

  const acceptKeybinding = $settings.suggestions.acceptKeybinding

  // Tab: accept suggestion
  if (event.key === 'Tab' && acceptKeybinding === 'tab') {
    event.preventDefault()

    if (event.shiftKey) {
      // Shift+Tab: toggle current line acceptance (for line-by-line UI)
      const currentLine = editor.state.selection.main.head  // Simplified: would need actual line number
      // toggleLineAcceptance effect dispatch
    } else {
      // Tab: commit all accepted lines
      commitSuggestion(editor, suggestion)
    }
  } else if (event.key === 'Escape') {
    // Escape: reject suggestion
    event.preventDefault()
    rejectSuggestion(editor, suggestion)
  }
}

function commitSuggestion(editor: any, suggestion: Suggestion) {
  const acceptedLines = getAcceptedLines(editor)
  const accepted = suggestion.lines
    .filter((_, i) => acceptedLines.size === 0 || acceptedLines.has(i))
    .join('\n')

  // Insert via transaction (preserves undo/redo per Phase 9 RESEARCH.md Pattern 2)
  const cursorPos = editor.state.selection.main.head
  editor.dispatch({
    changes: {
      from: cursorPos,
      to: cursorPos,
      insert: accepted
    }
  })

  // Record telemetry (Wave 1: services)
  suggestionService?.recordAcceptanceTelemetry(
    $selectedDocumentId,
    suggestion.provider,
    suggestion.lines.length,
    acceptedLines.size || suggestion.lines.length
  )

  // Clear suggestion decoration
  clearSuggestion(editor)
}

function rejectSuggestion(editor: any, suggestion: Suggestion) {
  suggestionService?.recordRejectionTelemetry(
    $selectedDocumentId,
    suggestion.provider
  )

  clearSuggestion(editor)
}

// Cleanup on unmount
$effect.pre(() => {
  return () => {
    suggestionService?.cancel()
  }
})
