    **State and logic (in <script>):**
    ```typescript
    let layoutMode: 'sectioned' | 'unified' = $state('sectioned')
    let sectionsCollapsed = $state({ papers: false, ideas: false, completions: false })
    let papers = $state<SidebarSuggestionItem[]>([])     // Local reactive view
    let ideas = $state<SidebarSuggestionItem[]>([])
    let completions = $state<SidebarSuggestionItem[]>([])
    let paperMatcher = $state<PaperKeywordMatcher | null>(null)  // Created in $effect
    let boundaryDetector = $state<SectionBoundaryDetector | null>(null)

    // Props
    let { docId, content = '', cursorPos = 0, suggestionService = null }: Props = $props()

    // Sync local state from global store
    $effect(() => {
      papers = sidebarPapers.current    // Reactive: updates when store changes
      ideas = sidebarIdeas.current
      completions = sidebarCompletions.current
    })

    // Continuous paper matching on content change (D-05 Tier 1 — no LLM)
    $effect(() => {
      const c = content
      if (!c || c.length < 10 || !paperMatcher) return
      const allPapers = get(papers)             // get() = svelte/store get
      const matches = paperMatcher.match(c, allPapers)
      setPaperMatches(matches)
    })

    // Initialize services
    $effect(() => {
      paperMatcher = new PaperKeywordMatcher()

      boundaryDetector = new SectionBoundaryDetector(
        async (event) => {
          if (!suggestionService) return
          setGenerating(true)
          try {
            const [ideasResult, completionsResult] = await Promise.all([
              suggestionService.generateIdeas(event.docId, event.content, event.cursorPos),
              suggestionService.generateCompletions(event.docId, event.content, event.cursorPos),
            ])
            setIdeas(ideasResult)
            setCompletions(completionsResult)
          } catch (err) {
            console.error('[SuggestionsSidebar] Generation failed:', err)
          } finally {
            setGenerating(false)
          }
        },
        () => content,                           // getContent callback
        () => docId                              // getDocId callback
      )

      return () => {
        boundaryDetector?.cleanup()
        clearAll()
      }
    })

    // Feed content changes to boundary detector
    $effect(() => {
      const c = content; const pos = cursorPos
      if (boundaryDetector) boundaryDetector.onContentChange(pos)
    })

    // Manual suggest button handler (D-04)
    function handleSuggest() {
      if (boundaryDetector) boundaryDetector.triggerManual(cursorPos)
    }

    // Paper click: insert citation (D-06 — placeholder, wired in Plan 03)
    function handlePaperClick(item: SidebarSuggestionItem) {
      if (!item.paper) return
      // Plan 03 will wire SmartCitationInsert here
      console.log('[SuggestionsSidebar] Paper clicked:', item.paper.title)
      // For now: dispatch a custom event that DocumentEditor listens for
      const event = new CustomEvent('suggestion-insert', {
        detail: { type: 'paper', item }
      })
      window.dispatchEvent(event)
    }

    // Completion click: insert text
    function handleCompletionClick(item: SidebarSuggestionItem) {
      const event = new CustomEvent('suggestion-insert', {
        detail: { type: 'completion', item }
      })
      window.dispatchEvent(event)
    }
    ```
