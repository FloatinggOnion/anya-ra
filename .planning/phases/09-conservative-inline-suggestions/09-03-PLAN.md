    ```typescript
    import type { Paper } from '../types/paper'
    import type { CitationStyle, SidebarSuggestionItem } from '../types/suggestion'
    import { get } from 'svelte/store'
    import { settings } from '../stores/settings'

    /**
     * Pool of framing phrases for contextual citations (D-06).
     * Rotated per paper ID to provide variety for repeated citations of the same paper.
     * The user can edit the framing text after insertion.
     */
    const FRAMING_PHRASES = [
      'As demonstrated by',
      'As noted by',
      'Building on work by',
      'As shown by',
      'According to',
      'As argued by',
      'In the work of',
      'As established by',
    ]

    /** Track which framing phrase index each paper last used, for rotation */
    const _paperFrameIndex = new Map<string, number>()

    /** Get the next framing phrase for a given paper. Cycles through the pool. */
    function getFramingText(paperId: string): string {
      const idx = _paperFrameIndex.get(paperId) ?? 0
      const phrase = FRAMING_PHRASES[idx % FRAMING_PHRASES.length]
      _paperFrameIndex.set(paperId, idx + 1)
      return phrase
    }

    /** Reset framing state. Exported for test isolation. */
    export function _resetFramingState(): void {
      _paperFrameIndex.clear()
    }

    /**
     * Generate contextual citation text based on the configured citation style.
     * Used when user clicks a suggestion card in the sidebar.
     *
     * @param item - The sidebar suggestion item clicked
     * @param paper - Optional full paper object (required for paper-type items)
     * @returns String to insert at cursor position in document
     */
    export function generateCitation(
      item: SidebarSuggestionItem,
      paper?: Paper | null
    ): string {
      // For text completions: just return the content with trailing space
      if (item.type === 'completion') {
        return item.content + ' '
      }

      // For ideas: no insertion — ideas are informational only
      if (item.type === 'idea') {
        return ''  // Ideas don't insert text
      }

      // For paper suggestions: generate contextual citation
      if (item.type === 'paper' && paper) {
        const style = getCitationStyle()
        return formatPaperCitation(paper, style)
      }

      // For paper suggestions without paper data: use item.paper info
      if (item.type === 'paper' && item.paper) {
        const style = getCitationStyle()
        // Build a minimal paper object from item.paper
        const minimalPaper: Paper = {
          id: item.paper.id,
          source: 'local' as const,
          externalId: item.paper.id,
          title: item.paper.title,
          authors: item.paper.authors,
          year: item.paper.year,
          abstract: null,
          url: '',
          doi: null,
          arxivId: null,
          semanticId: null,
          isOpenAccess: false,
          openAccessStatus: null,
          pdfUrl: null,
          localPdfPath: null,
          pdfDownloaded: false,
          importedAt: new Date().toISOString(),
          addedAt: new Date().toISOString(),
          lastUpdated: new Date().toISOString(),
          tags: [],
        }
        return formatPaperCitation(minimalPaper, style)
      }

      return ''
    }

    /**
     * Format a paper citation according to the given style.
     * PER D-06: "inserts a contextual citation with surrounding framing text"
     */
    function formatPaperCitation(paper: Paper, style: CitationStyle): string {
      const authors = paper.authors
      const title = paper.title
      const year = paper.year
      const yearStr = year ? ` (${year})` : ''

      switch (style) {
        case 'author-year': {
          const authorStr = formatAuthorName(authors)
          const frame = getFramingText(paper.id)
          // D-06: "contextual citation with surrounding framing text"
          // Produces e.g. "As demonstrated by Vaswani and Shazeer (2017) [cite: Attention Is All You Need], "
          return `${frame} ${authorStr}${yearStr} [cite: ${title}], `
        }

        case 'cite-title': {
          // Just the cite tag with surrounding spaces
          return ` [cite: ${title}] `
        }

        case 'footnote': {
          // Footnote style: [^N]: Authors, "Title" (Year)
          // The N should be dynamically determined, but since we can't easily
          // scan the document for existing footnote numbers from a service,
          // we use a placeholder notation that the user can adjust.
          // Better approach: insert a footnote-style reference inline.
          const authorStr = authors.slice(0, 3).map(a => {
            const parts = a.split(' ')
            return parts[parts.length - 1]  // Last name
          }).join(', ')
          const fullAuthorStr = authors.length > 3 ? `${authorStr} et al.` : authorStr
          const quote = `"${title}"`
          return `${fullAuthorStr}${yearStr}, ${quote}. `
        }

        default:
          return ` [cite: ${title}] `
      }
    }

    /**
     * Format author names for inline citation.
     * 1 author: "Smith"
     * 2 authors: "Smith and Jones"
     * 3+ authors: "Smith et al."
     */
    function formatAuthorName(authors: string[]): string {
      if (authors.length === 0) return 'Unknown'

      const surnames = authors.map(a => {
        const parts = a.trim().split(' ')
        return parts[parts.length - 1]
      })

      if (surnames.length === 1) return surnames[0]
      if (surnames.length === 2) return `${surnames[0]} and ${surnames[1]}`
      return `${surnames[0]} et al.`
    }

    /** Get the current citation style from the settings store */
    export function getCitationStyle(): CitationStyle {
      return get(settings).suggestions.citationStyle || 'author-year'
    }
    ```

    ```typescript
    import type { SidebarSuggestionItem } from '../types/suggestion'

    /**
     * Record a sidebar click-to-insert event.
     * Tracks: docId, suggestion type (paper/completion), provider, timestamp.
     * Silently ignores if telemetry opt-in disabled (D-11).
     */
    export function recordInsertion(
      docId: string,
      provider: string,          // 'paper-keyword' | 'ollama' | 'openrouter'
      suggestionType: string     // 'paper' | 'completion'
    ): void {
      const optIn = get(settings).suggestions.telemetryOptIn
      if (!optIn) return

      telemetryBuffer.push({
        docId,
        provider: provider === 'paper-keyword' ? 'ollama' : (provider as 'ollama' | 'openrouter'),
        type: 'accept',          // Click-to-insert is always a full acceptance in sidebar model
        lineCount: 1,            // Single insertion per click
        timestamp: Date.now()
      })

      if (telemetryBuffer.length % 10 === 0) {
        persistTelemetry()
      }
    }
    ```
