    Implementation details for `match()`:
    1. Call `extractKeywords(content)` → get keyword array
    2. If keywords.length === 0, return []
    3. For each paper in papers: call `scorePaper(paper, keywords)` → get score
    4. Filter scores > 0.05 (noise floor — papers with no real match get 0)
    5. Sort by relevanceScore descending
    6. Take top MAX_RESULTS (5)
    7. Return as PaperMatch[] with matchedKeywords (which keywords scored > 0)

    Implementation details for `scorePaper()`:
    1. For each keyword, check if it appears in paper.title (case-insensitive includes) → weight 3.0 per match
    2. Check paper.authors (any author contains keyword, case-insensitive) → weight 2.0 per match
    3. Check paper.abstract (if non-null, case-insensitive includes) → weight 1.0 per match
    4. Total raw score = sum of weighted matches
    5. Normalize: rawScore / (keywords.length × 6) then clamp to [0, 1]
    6. Return normalized score

    /**
     * Generate ideas/questions for the sidebar (D-02, D-04).
     * Uses LLM via provider chain. Triggered on section boundary or manual button.
     *
     * @returns Array of SidebarSuggestionItem (type: 'idea'), max 3
     */
    async generateIdeas(docId: string, currentContent: string, cursorPos: number): Promise<SidebarSuggestionItem[]> {
      const context = this._buildContext(currentContent, cursorPos)
      const messages: Message[] = [
        {
          role: 'system',
          content: 'You are a research writing assistant. Based on the document context, generate 1-3 brief ideas, counterarguments, or follow-up questions that could strengthen this writing. Each idea should be one sentence. Format: one idea per line, prefixed with "• " (bullet). No numbering. No explanations. No more than 3 lines total.'
        },
        { role: 'user', content: context }
      ]

      const settingsVal = get(settings)
      const model = settingsVal.suggestions.llmProvider === 'ollama'
        ? settingsVal.suggestions.ollamaModel
        : settingsVal.suggestions.openrouterModel

      const text = await this._streamSuggestion(messages, new AbortController().signal, model)
      return this._parseIdeas(text, docId)
    }

    /**
     * Generate text completions for the sidebar (D-02, D-04).
     * Uses LLM via provider chain. Triggered on section boundary or manual button.
     *
     * @returns Array of SidebarSuggestionItem (type: 'completion'), max 2
     */
    async generateCompletions(docId: string, currentContent: string, cursorPos: number): Promise<SidebarSuggestionItem[]> {
      const context = this._buildContext(currentContent, cursorPos)
      const messages: Message[] = [
        {
          role: 'system',
          content: 'You are a writing assistant helping a researcher. Based on the document context, generate exactly 1-2 brief sentence continuations that logically follow the last written text. Each completion should be one sentence. Format: one completion per line, plain text only (no prefixes, no numbering). No more than 2 lines. No explanations.'
        },
        { role: 'user', content: context }
      ]

      const settingsVal = get(settings)
      const model = settingsVal.suggestions.llmProvider === 'ollama'
        ? settingsVal.suggestions.ollamaModel
        : settingsVal.suggestions.openrouterModel

      const text = await this._streamSuggestion(messages, new AbortController().signal, model)
      return this._parseCompletions(text, docId)
    }

    /** Parse ideas text response into SidebarSuggestionItem[] */
    private _parseIdeas(text: string, docId: string): SidebarSuggestionItem[] {
      return text.split('\n')
        .filter(line => line.trim().startsWith('•'))
        .map((line, i) => ({
          id: `idea-${docId}-${Date.now()}-${i}`,
          type: 'idea' as const,
          label: 'Ideas',
          content: line.replace(/^•\s*/, '').trim(),
          timestamp: Date.now(),
          source: 'llm' as const,
        }))
        .slice(0, 3)
    }

    /** Parse completions text response into SidebarSuggestionItem[] */
    private _parseCompletions(text: string, docId: string): SidebarSuggestionItem[] {
      return text.split('\n')
        .filter(line => line.trim().length > 0)
        .map((line, i) => ({
          id: `completion-${docId}-${Date.now()}-${i}`,
          type: 'completion' as const,
          label: 'Completions',
          content: line.trim(),
          timestamp: Date.now(),
          source: 'llm' as const,
          provider: get(settings).suggestions.llmProvider,
        }))
        .slice(0, 2)
    }
